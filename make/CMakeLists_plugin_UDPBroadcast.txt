# plugin CMake file to be included in main project CMakelists

# --- UDPBroadcast plugin ---

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD 99)

set(UDPBROADCAST_PLUGIN_SOURCES
   plugins/udp-broadcast/udp_broadcast_plugin.cpp
   plugins/udp-broadcast/event_collector.cpp
   plugins/udp-broadcast/broadcaster.cpp
   plugins/udp-broadcast/system.cpp
)

if(BUILD_SHARED)
   add_library(UDPBroadcastPlugin MODULE ${UDPBROADCAST_PLUGIN_SOURCES})

   target_include_directories(UDPBroadcastPlugin PUBLIC
      plugins/plugins/udp-broadcast
      ${CMAKE_SOURCE_DIR}/third-party/include
      plugins/plugins
      src
   )

   if(PluginPlatform STREQUAL "windows")
      target_link_libraries(UDPBroadcastPlugin PRIVATE ws2_32)

      add_compile_options(
         $<$<CONFIG:RELEASE>:/Ob2>
         $<$<CONFIG:RELEASE>:/O2>
         $<$<CONFIG:RELEASE>:/Oi>
         $<$<CONFIG:RELEASE>:/arch:SSE2>
         $<$<CONFIG:RELEASE>:/fp:fast>
         $<$<CONFIG:RELEASE>:/fp:except->
         $<$<CONFIG:RELEASE>:/Ot>
         $<$<CONFIG:RELEASE>:/GF>
         $<$<CONFIG:RELEASE>:/GS->
         $<$<CONFIG:RELEASE>:/Gy>
         $<$<CONFIG:RELEASE>:/GR>
         $<$<CONFIG:RELEASE>:/Oy>
         $<$<CONFIG:RELEASE>:/GT>
         $<$<CONFIG:RELEASE>:/GL>
      )
      set_target_properties(UDPBroadcastPlugin PROPERTIES
         MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:DEBUG>:Debug>"
      )
      target_compile_options(UDPBroadcastPlugin PUBLIC
         $<$<CONFIG:RELEASE>:$<$<COMPILE_LANGUAGE:CXX>:${OPT_COMMON}>>
         $<$<CONFIG:RELEASE>:$<$<COMPILE_LANGUAGE:C>:${OPT_COMMON}>>
      )
      target_link_options(UDPBroadcastPlugin PUBLIC
         $<$<CONFIG:RELEASE>:/INCREMENTAL:NO>
         $<$<CONFIG:RELEASE>:/OPT:REF>
         $<$<CONFIG:RELEASE>:/OPT:ICF>
         $<$<CONFIG:RELEASE>:/LTCG>
      )
      if (PluginArch STREQUAL "x64")
         set_target_properties(UDPBroadcastPlugin PROPERTIES
            OUTPUT_NAME "plugin-udp-broadcast64"
         )
      else()
         set_target_properties(UDPBroadcastPlugin PROPERTIES
            OUTPUT_NAME "plugin-udp-broadcast"
         )
      endif()
   else()
      # Link with pthread on Linux/macOS
      target_link_libraries(UDPBroadcastPlugin PRIVATE pthread)

      set_target_properties(UDPBroadcastPlugin PROPERTIES
         PREFIX ""
         OUTPUT_NAME "plugin-udp-broadcast"
      )
      if(PluginPlatform STREQUAL "macos")
         set_target_properties(UDPBroadcastPlugin PROPERTIES
            SUFFIX ".dylib"
         )
      endif()
   endif()

   add_custom_command(TARGET UDPBroadcastPlugin POST_BUILD
      COMMAND "${CMAKE_COMMAND}" -E copy "${CMAKE_SOURCE_DIR}/plugins/udp-broadcast/plugin.cfg" "${PLUGINS_DIR}/udp-broadcast/plugin.cfg"
      COMMAND "${CMAKE_COMMAND}" -E copy "$<TARGET_FILE:UDPBroadcastPlugin>" "${PLUGINS_DIR}/udp-broadcast"
      COMMAND "${CMAKE_COMMAND}" -E rm "$<TARGET_FILE:UDPBroadcastPlugin>"
   )
endif()

if(BUILD_STATIC)
   add_compile_options(-fPIC)

   add_library(UDPBroadcastPlugin STATIC ${UDPBROADCAST_PLUGIN_SOURCES})

   target_include_directories(UDPBroadcastPlugin PUBLIC
      plugins/plugins/udp-broadcast
      ${CMAKE_SOURCE_DIR}/third-party/include
      plugins/plugins
      src
   )

   # Link with pthread on non-Windows
   if(NOT PluginPlatform STREQUAL "windows")
      target_link_libraries(UDPBroadcastPlugin PRIVATE pthread)
   else()
      target_link_libraries(UDPBroadcastPlugin PRIVATE ws2_32)
   endif()

   set_target_properties(UDPBroadcastPlugin PROPERTIES
      PREFIX ""
      OUTPUT_NAME "plugin-udp-broadcast"
   )
endif()
